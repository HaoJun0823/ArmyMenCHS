{
	IF PICKEDUP private_@1 BY captain TESTVAR GameStage = 3 AND
		HASITEM captain private_@2 = 1 AND
		HASITEM captain private_@3 = 1 AND
		HASITEM captain private_@4 = 1 AND
		HASITEM captain private_@5 = 1 THEN

			SETVAR GameStage 4,
			SETVAR HelpBeacon 10,
			SETVAR HelpLanding 0,

			PLAYTALKINGHEAD 127,

}
