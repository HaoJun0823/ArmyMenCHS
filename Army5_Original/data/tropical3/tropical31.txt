{

			TRIGGER selectsubweapon lastcreated,
			TRIGGER randsublocation lastcreated
	}
}

{
	UNTIL @2 REPEAT 1 BY 1
	{
		IF createchoppers TESTVAR Difficulty = @1 AND BoolWarning = 0 THEN
			CREATECOPTER scout tanscout_$ (100,100,200) tan,

		IF createchoppers TESTVAR Difficulty = @1 AND BoolWarning > 0 THEN
