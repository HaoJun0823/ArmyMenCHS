MACRO createsubsMacro 2
{
	IF createsubs TESTVAR Difficulty = @1 AND NumGreySubs < @2 THEN
		ADDVAR NumGreySubs 2,
		CREATEUNIT gsub submarine padSub26 grey,
		SELECTINVENTORYITEM lastcreated 2,
		TRIGGER ordersub26 lastcreated,
		CREATEUNIT gsub submarine padSub27 grey,
		SELECTINVENTORYITEM lastcreated 2,
		TRIGGER ordersub27 lastcreated,
		TRIGGERDELAY 10000 createsubs

	IF createsubs TESTVAR Difficulty = @1 AND NumGreySubs >= @2 THEN
		TRIGGERDELAY 1000 createsubs
}
