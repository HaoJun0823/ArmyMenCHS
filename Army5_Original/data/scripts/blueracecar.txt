MACRO moveBlueCar 8 
{
	IF moveblue_@1 THEN
		TRIGGER setupmovebluestate_@1,
		SETVAR targetXblue sourceXblue,
		ADDVAR targetXblue @3,
		SETVAR targetYblue sourceYblue,
		ADDVAR targetYblue @4,
		SETVAR currentZBlue @8,
		TRIGGER movebluedist_@1,
		TRIGGER testbluemove_@1

	IF setupmovebluestate_@1 TESTVAR newbluestate = 0 THEN
		SETVAR newbluestate 1,
		SETVAR sourceXblue startXblue, ADDVAR sourceXblue @6,
		SETVAR sourceYblue startYblue, ADDVAR sourceYblue @7
	
	IF movebluedist_@1 THEN
		ADDVAR tallyXblue @2,
		SETVAR currentXblue sourceXblue,
		ADDVAR currentXblue tallyXblue,
		SETVAR currentZBlue @8,

		SETVAR tallyYblue tallyXblue,
		MULVAR tallyYblue @4,
		DIVVAR tallyYblue @3,
		SETVAR currentYblue sourceYblue,
		ADDVAR currentYblue tallyYblue

	IF testbluemove_@1 TESTVAR @2 > 0 THEN
		TRIGGER posbluemove_@1

	IF testbluemove_@1 TESTVAR @2 < 0 THEN
		TRIGGER negbluemove_@1

	IF posbluemove_@1 TESTVAR currentXblue < targetXblue THEN
		MOVEITEM BlueCar (REFVAR currentXblue, REFVAR currentYblue, REFVAR currentZBlue)

	IF posbluemove_@1 TESTVAR currentXblue >= targetXblue THEN
		SETVAR newbluestate 0,
		SETOBJSTATE BlueCar osBlue@5,
		MOVEITEM BlueCar (REFVAR currentXblue, REFVAR currentYblue,REFVAR currentZBlue)

	IF negbluemove_@1 TESTVAR currentXblue > targetXblue THEN
		MOVEITEM BlueCar (REFVAR currentXblue, REFVAR currentYblue,REFVAR currentZBlue)

	IF negbluemove_@1 TESTVAR currentXblue <= targetXblue THEN
		SETVAR newbluestate 0,
		SETOBJSTATE BlueCar osBlue@5,
		MOVEITEM BlueCar (REFVAR currentXblue, REFVAR currentYblue,REFVAR currentZBlue)
}

MACRO arcBlueCar 7
{
	IF arcblue_@1 THEN
		TRIGGER setuparcbluestate_@1,
		SETVAR enddegblue @3,
		SETVAR centerXblue startXblue,
		ADDVAR centerXblue @5,
		SETVAR centerYblue startYblue,
		ADDVAR centerYblue @6,
		TRIGGER testbluedist_@1,
		TRIGGER arcbluedist_@1,
		TRIGGER testbluearc_@1
	
	IF setuparcbluestate_@1 TESTVAR newbluestate = 0 THEN
		SETVAR newbluestate 1,
		SETVAR currentdegblue @2,
		SETVAR tallyXblue 0, SETVAR tallyYblue 0

	IF testbluedist_@1 TESTVAR @4 > 0 THEN
		//SETVAR rXblue 380, SETVAR rYblue 314
		SETVAR rXblue 390, SETVAR rYblue 324

	IF testbluedist_@1 TESTVAR @4 < 0 THEN
		SETVAR rXblue 330, SETVAR rYblue 274

	IF arcbluedist_@1 THEN
		ADDVAR currentdegblue @4,
		SETVAR cosblue currentdegblue,
		SETVAR sinblue currentdegblue,
		COSVAR cosblue, SINVAR sinblue,
		
		SETVAR currentXblue rXblue,
		MULVAR currentXblue cosblue,
		DIVVAR currentXblue 1000,
		ADDVAR currentXblue centerXblue,
	
		SETVAR currentYblue rYblue,
		MULVAR currentYblue sinblue,
		DIVVAR currentYblue 1000,
		ADDVAR currentYblue centerYblue
	
	IF testbluearc_@1 TESTVAR @4 > 0 THEN
		TRIGGER posbluearc_@1

	IF testbluearc_@1 TESTVAR @4 < 0 THEN
		TRIGGER negbluearc_@1

	IF posbluearc_@1 TESTVAR currentdegblue < enddegblue THEN
		MOVEITEM BlueCar (REFVAR currentXblue, REFVAR currentYblue)

	IF posbluearc_@1 TESTVAR currentdegblue >= enddegblue THEN
		SETVAR newbluestate 0,
		SETOBJSTATE BlueCar osBlue@7,
		MOVEITEM BlueCar (REFVAR currentXblue, REFVAR currentYblue)

	IF negbluearc_@1 TESTVAR currentdegblue > enddegblue THEN
		MOVEITEM BlueCar (REFVAR currentXblue, REFVAR currentYblue)

	IF negbluearc_@1 TESTVAR currentdegblue <= enddegblue THEN
		SETVAR newbluestate 0,
		SETOBJSTATE BlueCar osBlue@7,
		MOVEITEM BlueCar (REFVAR currentXblue, REFVAR currentYblue)
}
