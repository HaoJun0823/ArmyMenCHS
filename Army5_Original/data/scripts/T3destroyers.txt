//*** SETUP GUNS ***//

	SETITEMOWNER Destroyer2 green,

	CREATEUNIT D1gun2 DESTROYERROCKETS (1,1) green, TRIGGER D1gun2spot,
	CREATEUNIT D1gun3 DESTROYERRADAR (1,1) green, TRIGGER D1gun3spot,
	CREATEUNIT D1gun4 DESTROYERROCKETS (1,1) green, TRIGGER D1gun4spot,
	CREATEUNIT D1gun5 DESTROYERGUN (1,1) green, TRIGGER D1gun5spot,
	CREATEUNIT D2gun1 DESTROYERGUN (1,1) green, TRIGGER D2gun1spot,
	CREATEUNIT D2gun2 DESTROYERROCKETS (1,1) green, TRIGGER D2gun2spot,
	CREATEUNIT D2gun3 DESTROYERRADAR (1,1) green, TRIGGER D2gun3spot,
	CREATEUNIT D2gun4 DESTROYERROCKETS (1,1) green, TRIGGER D2gun4spot,
	CREATEUNIT D2gun5 DESTROYERGUN (1,1) green, TRIGGER D2gun5spot

{
	IF setupdestroyerradar THEN
		RADARGUARDS D1gun3 D1gun$,
		RADARGUARDS D2gun3 D2gun$
}

MACRO destroyergunlocations 6
{
	IF D@1gun@2spot THEN
		SETVAR D1gunX GETX Destroyer@1, SETVAR D1gunY GETY Destroyer@1,
		ADDVAR D1gunX @3, ADDVAR D1gunY @4,
		MOVEITEM D@1gun@2 (REFVAR D1gunX, REFVAR D1gunY, @5),
		SETFACING D@1gun@2 @6
} 
EXPAND destroyergunlocations ( 1 1 -56  0 10  32 )
EXPAND destroyergunlocations ( 1 2  13  0 10  16 )
EXPAND destroyergunlocations ( 1 3  75 -5 10   0 )
EXPAND destroyergunlocations ( 1 4 146 15 10 240 )
EXPAND destroyergunlocations ( 1 5 211 15 10 224 )
EXPAND destroyergunlocations ( 2 1 -56  0 10  96 )
EXPAND destroyergunlocations ( 2 2  13  0 10 112 )
EXPAND destroyergunlocations ( 2 3  75 -5 10 128 )
EXPAND destroyergunlocations ( 2 4 146 15 10 144 )
EXPAND destroyergunlocations ( 2 5 211 15 10 160 )

//*** SETUP GUNS ***//


//*** DESTROYER DAMAGE ***//

UNTIL 5 REPEAT 1 BY 1
{
	IF removeallguns THEN
		UNDEPLOY D1gun$,
		UNDEPLOY D2gun$
}

UNTIL 2 REPEAT 1 BY 1
{
	IF KILLED D$gunLIST THEN
		CREATEEXPLOSION me 143,
		ADDVAR dest$gunsleft -1,
		TRIGGER checkfordeadD$

	IF checkfordeadD$ TESTVAR dest$gunsleft <= 0 THEN
		SETCAMERAFOCUS Destroyer$,
		TRIGGERDELAY 1000 destroyerkilled Destroyer$
}

//*** DESTROYER DAMAGE ***//


//*** CARRIER LZ ***//

UNTIL 2 REPEAT 1 BY 1
{
	IF PADON padDest$ TESTVAR carrierLZ = 1 THEN
		TRIGGERDELAY 500 copterlanding$

	IF copterlanding$ TESTVAR GETPADSTATUS padDest$ = 1 AND carrierhaslanded = 0 THEN
		LAND captain,
		TRIGGERDELAY 1 copterlanding$

	IF LANDED captain TESTVAR GETPADSTATUS padDest$ = 1 THEN
		SETVAR carrierhaslanded 1

	IF TOOKOFF captain THEN
		SETVAR carrierhaslanded 0
}

//*** CARRIER LZ ***//


//*** GREEN DESTROYERS ***//

IF setupdestroyers THEN
	SETVAR carrierLZ 1,
	SETOBJSTATE Destroyer1 osMove, SETOBJSTATE Destroyer2 osMove,
	TRIGGER setupdestroyerguns,
	TRIGGER setupdestroyerradar
