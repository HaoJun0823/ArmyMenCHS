// PLAYER LZ and RESPAWN ======================

MACRO playerrespawn 1
{
	IF PADON pad@1LZ THEN
		TRIGGERDELAY 1000 @1_copterlanding

	IF @1_copterlanding TESTVAR GETPADSTATUS pad@1LZ = 1 AND @1Eaglehaslanded = 0 THEN
		LAND captain_@1,
		TRIGGERDELAY 100 @1_copterlanding

	IF LANDED captain_@1 THEN
		SETVAR @1Eaglehaslanded 1

	IF TOOKOFF captain_@1 THEN
		SETVAR @1Eaglehaslanded 0

	IF KILLED captain_@1 THEN
		SETVAR @1Eaglehaslanded 0,
		TRIGGERDELAY REFVAR RespawnTimer @1_bringoutyourdead
	
	IF @1_bringoutyourdead THEN
		TRIGGER @1_respawn,
		SETVAR captainX GETX captain_@1, SETVAR captainY GETY captain_@1,
		MOVEITEM captain_@1 (REFVAR captainX, REFVAR captainY, 0)

	IF @1_respawn THEN
		DEPLOY captain_@1 pad@1LZ
}
EXPAND playerrespawn ( Green )
EXPAND playerrespawn ( Tan )
EXPAND playerrespawn ( Grey )
EXPAND playerrespawn ( Blue )

MACRO EWSHandler 4
{
	IF PADON pad@2At@1EWS TESTVAR boolEWSActive = 1 AND bool@1EWSDelay = 0 THEN
		SHOWMESSAGE "@2 unit detected near @1 landing zone.",
		TRIGGERDELAY 5 @1_setEWSdelay

	IF PADON pad@3At@1EWS TESTVAR boolEWSActive = 1 AND bool@1EWSDelay = 0 THEN
		SHOWMESSAGE "@3 unit detected near @1 landing zone.",
		TRIGGERDELAY 5  @1_setEWSdelay

	IF PADON pad@4At@1EWS TESTVAR boolEWSActive = 1 AND bool@1EWSDelay = 0 THEN
		SHOWMESSAGE "@4 unit detected near @1 landing zone.",
		TRIGGERDELAY 5 @1_setEWSdelay

	IF @1_setEWSdelay THEN
		SETVAR bool@1EWSDelay 1,
		TRIGGERDELAY REFVAR EWSDelayTimer @1_clearEWSbuffer

	IF @1_clearEWSbuffer THEN
		SETVAR bool@1EWSDelay 0
}
EXPAND EWSHandler ( "Tan" "Blue" "Green" "Grey" )
EXPAND EWSHandler ( "Green" "Blue" "Tan" "Grey" )
EXPAND EWSHandler ( "Blue" "Green" "Tan" "Grey" )

MACRO KillerFragDude 2
{
	IF KILLED @1 @2 BY @1 THEN
		MULVAR @2points -2,
		ADDVAR @1score @2points,
		DIVVAR @2points -2

	IF KILLED @2 BY @1 THEN
		ADDVAR @1score @2points,
		TRIGGERDELAY 1 checkteamvictory
}

MACRO CopterFrag 4
{
	IF KILLED captain_@1 BY captain_@1 THEN
	    SHOWMESSAGE "took a self-inflicted fatality!" @1

	IF KILLED captain_@1 BY @1 THEN 
		MULVAR copterpoints -2,
		ADDVAR @1score copterpoints,
		DIVVAR copterpoints -2

	IF KILLED captain_@2 BY captain_@1 TESTVAR ISALLY @1 @2 = 0 THEN
	    SHOWMESSAGE "was downed by" @2 @1

	IF KILLED captain_@2 BY @1 TESTVAR ISALLY @1 @2 = 0 THEN // reward player for kills made by his ground units
		ADDVAR @1score copterpoints,
		TRIGGERDELAY 1 checkteamvictory

	IF KILLED captain_@2 BY captain_@1 TESTVAR ISALLY @1 @2 = 1 THEN
	    SHOWMESSAGE "killed in friendly fire by" @2 @1

	IF KILLED captain_@2 BY @1 TESTVAR ISALLY @1 @2 = 1 THEN
		ADDVAR @1score allycopterpoints

	IF KILLED captain_@3 BY captain_@1 TESTVAR ISALLY @1 @3 = 0 THEN
	    SHOWMESSAGE "was downed by" @3 @1

	IF KILLED captain_@3 BY @1 TESTVAR ISALLY @1 @3 = 0 THEN
		ADDVAR @1score copterpoints,
		TRIGGERDELAY 1 checkteamvictory

	IF KILLED captain_@3 BY captain_@1 TESTVAR ISALLY @1 @3 = 1 THEN
	    SHOWMESSAGE "killed in friendly fire by" @3 @1

	IF KILLED captain_@3 BY @1 TESTVAR ISALLY @1 @3 = 1 THEN
		ADDVAR @1score allycopterpoints

	IF KILLED captain_@4 BY captain_@1 TESTVAR ISALLY @1 @4 = 0 THEN
	    SHOWMESSAGE "was downed by" @4 @1

	IF KILLED captain_@4 BY @1 TESTVAR ISALLY @1 @4 = 0 THEN
		ADDVAR @1score copterpoints,
		TRIGGERDELAY 1 checkteamvictory

	IF KILLED captain_@4 BY captain_@1 TESTVAR ISALLY @1 @4 = 1 THEN
	    SHOWMESSAGE "killed in friendly fire by" @4 @1

	IF KILLED captain_@4 BY @1 TESTVAR ISALLY @1 @4 = 1 THEN
		ADDVAR @1score allycopterpoints
}
EXPAND CopterFrag ( Green Tan Blue Grey )
EXPAND CopterFrag ( Grey Green Tan Blue )
EXPAND CopterFrag ( Blue Grey Green Tan )
EXPAND CopterFrag ( Tan Blue Grey Green )
