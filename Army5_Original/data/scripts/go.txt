/////////////////////// CPU moves

variable varmovecount 0

if blackmoves butnot evtplayerwhite // when player moves
	then trigger evtCPUmove,
	trigger evtResetTimer // start counting toward pass hint again

	if evtCPUmove then
	triggerdelay 2000 evtrespond

if whitemoves butnot evtplayerblack // when player moves
	then trigger evtCPUmove,
	trigger evtResetTimer

if evtrespond then
	makegomove,
	addvar varmovecount 2,
	triggerdelay 5000 evtTimeNudge

if blackpasses butnot evtplayerblack	// if pc passes then let cpu pass for player
	then trigger evtrespond

if whitepasses butnot evtplayerwhite	// if pc passes then let cpu pass for player
	then trigger evtrespond

if  blackpasses butnot evtplayerwhite	// if human passes, consider it a move
	then trigger blackmoves

if  whitepasses butnot evtplayerblack	// if human passes, consider it a move
	then trigger whitemoves

///////////////////////// pass hint

variable booltimerblock 0
variable vargotimer 0

if evtResetTimer then
	setvar vargotimer 0,
	setvar booltimerblock 1,
	triggerdelay 3400 evtUnblockTimer,
	triggerdelay 3500 evtTimerloop

if evtUnblocktimer then setvar booltimerblock 0

if evtTimerloop butnot killed goplayer
	testvar booltimerblock = 0
	then addvar vargotimer 3,
	triggerdelay 3000 evtTimerloop

	if evttimerloop butnot evtpasshint1 // play passhint after x secs of inactivity
		testvar vargotimer = 24
		AND itemtoitemdistance objgoboard captain < 550
		then trigger evtpasshint

			if evtpasshint butnot evtmissionlost
				testvar boolVOlockout = 0
				then
				trigger evtpasshint1

				if evtpasshint1 butnot humanwins then
					trigger evtpasshint2

					if evtpasshint2 butnot computerwins then
						trigger evtpasshint3

						if evtpasshint3 butnot tiegame then
							playtalkinghead 133, // Shoot into your bowl to pass, Captain.
							setvar booltimerblock 1

///////////////////////////////// time nudges

			if evtTimeNudge butnot evtmissionlost
				testvar boolVOlockout = 0 AND
				varmovecount = 62 // game nearing end
				then
				trigger evtTimeNudge3

				if evtTimeNudge3 butnot humanwins then
					trigger evtTimeNudge4

					if evtTimeNudge4 butnot computerwins then
						trigger evtTimeNudge5

						if evtTimeNudge5 butnot tiegame then
							playtalkinghead 11, // Captain, rem your obj
							triggerdelay 25000 evtTimeSpank

	if inorder evtTimeSpank evtpasshint3 talkingheaddone captain
		testvar itemtoitemdistance objgoboard captain < 550
		then triggerdelay 15000 evtTimeSpank2

	if inorder evtpasshint3 evtTimeSpank
		testvar itemtoitemdistance objgoboard captain < 550
		then trigger evtTimeSpank2

			if evtTimeSpank2 butnot evtmissionlost
				testvar boolVOlockout = 0
				then
				trigger evtTimeSpank3

				if evtTimeSpank3 butnot humanwins then
					trigger evtTimeSpank4

					if evtTimeSpank4 butnot computerwins then
						trigger evtTimeSpank5

						if evtTimeSpank5 butnot tiegame then
							playtalkinghead 158 //You've still got work to do...! Shoot into your bowl, Captain.
