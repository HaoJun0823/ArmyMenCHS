MACRO moveRedCar 8
{
	IF movered_@1 THEN
		TRIGGER setupmoveredstate_@1,
		SETVAR targetXred sourceXred,
		ADDVAR targetXred @3,
		SETVAR targetYred sourceYred,
		ADDVAR targetYred @4,
		SETVAR currentZred @8,
		TRIGGER movereddist_@1,
		TRIGGER testredmove_@1

	IF setupmoveredstate_@1 TESTVAR newredstate = 0 THEN
		SETVAR newredstate 1,
		SETVAR sourceXred startXred, ADDVAR sourceXred @6,
		SETVAR sourceYred startYred, ADDVAR sourceYred @7
	
	IF movereddist_@1 THEN
		ADDVAR tallyXred @2,
		SETVAR currentXred sourceXred,
		ADDVAR currentXred tallyXred,
		SETVAR currentZred @8,

		SETVAR tallyYred tallyXred,
		MULVAR tallyYred @4,
		DIVVAR tallyYred @3,
		SETVAR currentYred sourceYred,
		ADDVAR currentYred tallyYred

	IF testredmove_@1 TESTVAR @2 > 0 THEN
		TRIGGER posredmove_@1

	IF testredmove_@1 TESTVAR @2 < 0 THEN
		TRIGGER negredmove_@1

	IF posredmove_@1 TESTVAR currentXred < targetXred THEN
		MOVEITEM RedCar (REFVAR currentXred, REFVAR currentYred, REFVAR currentZred)

	IF posredmove_@1 TESTVAR currentXred >= targetXred THEN
		SETVAR newredstate 0,
		SETOBJSTATE RedCar osRed@5,
		MOVEITEM RedCar (REFVAR currentXred, REFVAR currentYred, REFVAR currentZred)

	IF negredmove_@1 TESTVAR currentXred > targetXred THEN
		MOVEITEM RedCar (REFVAR currentXred, REFVAR currentYred, REFVAR currentZred)

	IF negredmove_@1 TESTVAR currentXred <= targetXred THEN
		SETVAR newredstate 0,
		SETOBJSTATE RedCar osRed@5,
		MOVEITEM RedCar (REFVAR currentXred, REFVAR currentYred, REFVAR currentZred)
}

MACRO arcRedCar 7
{
	IF arcred_@1 THEN
		TRIGGER setuparcredstate_@1,
		SETVAR enddegred @3,
		SETVAR centerXred startXred,
		ADDVAR centerXred @5,
		SETVAR centerYred startYred,
		ADDVAR centerYred @6,
		TRIGGER testreddist_@1,
		TRIGGER arcreddist_@1,
		TRIGGER testredarc_@1
	
	IF setuparcredstate_@1 TESTVAR newredstate = 0 THEN
		SETVAR newredstate 1,
		SETVAR currentdegred @2,
		SETVAR tallyXred 0, SETVAR tallyYred 0

	IF testreddist_@1 TESTVAR @4 > 0 THEN
		SETVAR rXred 330, SETVAR rYred 274
		
	IF testreddist_@1 TESTVAR @4 < 0 THEN
		SETVAR rXred 380, SETVAR rYred 314

	IF arcreddist_@1 THEN
		ADDVAR currentdegred @4,
		SETVAR cosred currentdegred,
		SETVAR sinred currentdegred,
		COSVAR cosred, SINVAR sinred,
		
		SETVAR currentXred rXred,
		MULVAR currentXred cosred,
		DIVVAR currentXred 1000,
		ADDVAR currentXred centerXred,
	
		SETVAR currentYred rYred,
		MULVAR currentYred sinred,
		DIVVAR currentYred 1000,
		ADDVAR currentYred centerYred
	
	IF testredarc_@1 TESTVAR @4 > 0 THEN
		TRIGGER posredarc_@1

	IF testredarc_@1 TESTVAR @4 < 0 THEN
		TRIGGER negredarc_@1

	IF posredarc_@1 TESTVAR currentdegred < enddegred THEN
		MOVEITEM RedCar (REFVAR currentXred, REFVAR currentYred)

	IF posredarc_@1 TESTVAR currentdegred >= enddegred THEN
		SETVAR newredstate 0,
		SETOBJSTATE RedCar osRed@7,
		MOVEITEM RedCar (REFVAR currentXred, REFVAR currentYred)

	IF negredarc_@1 TESTVAR currentdegred > enddegred THEN
		MOVEITEM RedCar (REFVAR currentXred, REFVAR currentYred)

	IF negredarc_@1 TESTVAR currentdegred <= enddegred THEN
		SETVAR newredstate 0,
		SETOBJSTATE RedCar osRed@7,
		MOVEITEM RedCar (REFVAR currentXred, REFVAR currentYred)
}
