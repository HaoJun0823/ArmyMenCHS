MACRO macrotreedamage 4
{
	IF DMGLVLCHANGED treefx@1 THEN
		TRIGGER treefxexplosion@2,
		TRIGGERdelay 2 treefxpoof@3,
		TRIGGERDELAY 50 treefxpoof@4
}
