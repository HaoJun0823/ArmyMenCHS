{
	IF mpsetupplayers THEN
		CREATECOPTER scout captain_@1 pad@1LZ @1,
		SETAIRMAP captain_@1 1,
		SETAIRVARIANCE captain_@1 200,
		SETSTEP captain_@1 500
	
	UNTIL 5 REPEAT 1 BY 1
	{
		IF mpsetupplayers TESTVAR ISCOLORINGAME @1 = 1 THEN
			CREATEUNIT @1_basepatrol_$ stingerguy pad@1AA$ @1 stingermissile
	}
}

MACRO WinchableHandler 7
{
	UNTIL 3 REPEAT 1 BY 1
	{
		// Ex: captain_Green drops Greenobj1 on his landing pad.
		IF DROPPED @1obj$ BY captain_@1 TESTVAR GETPADSTATUS pad@1LZ = 1 THEN
			UNDEPLOY me,
			DEPLOY me pad@1source$

		IF relocate@1obj$ THEN
			ADDVAR @1score 1,
			REMOVEMISSIONOBJECTIVE @5obj$ @1,
			REMOVEMISSIONOBJECTIVE @6obj$ @1,
			REMOVEMISSIONOBJECTIVE @7obj$ @1,
			TRIGGERDELAY 1 checkteamvictory

		// Ex: captain_Green drops Tanobj1 on his landing pad and greentarget1 is vacant.
		IF DROPPED @5obj$ BY captain_@1 TESTVAR quad@1Target$ = 0 AND GETPADSTATUS pad@1dropzone$ = 1 THEN
			TRIGGER syncvarA_@1_$,
			TRIGGERDELAY 1 relocate@1obj$

			IF syncvarA_@1_$ THEN
				SETVAR quad@1Target$ 5


		// Ex: captain_Green drops Greyobj1 on his landing pad and greentarget1 is vacant.
		IF DROPPED @6obj$ BY captain_@1 TESTVAR quad@1Target$ = 0 AND GETPADSTATUS pad@1dropzone$ = 1 THEN
			TRIGGER syncvarB_@1_$,
			TRIGGERDELAY 1 relocate@1obj$

			IF syncvarB_@1_$ THEN
				SETVAR quad@1Target$ 6


		// Ex: captain_Green drops Blueobj1 on his landing pad and greentarget1 is vacant.
		IF DROPPED @7obj$ BY captain_@1 TESTVAR quad@1Target$ = 0 AND GETPADSTATUS pad@1dropzone$ = 1 THEN
			TRIGGER syncvarC_@1_$,
			TRIGGERDELAY 1 relocate@1obj$
		
			IF syncvarC_@1_$ THEN
				SETVAR quad@1Target$ 7

		/////////////////////////////////////////

		// Ex: captain_Tan owns Greenobj1, then Greenobj1 is picked up.
		IF PICKEDUP @1obj$ TESTVAR quad@5Target$ = 7 THEN
			TRIGGER syncclientA_@1_@5_@6_@7_$

			IF syncclientA_@1_@5_@6_@7_$ THEN
				ADDVAR @5score -1,
				ADDMISSIONOBJECTIVE @1obj$ @5, ADDMISSIONOBJECTIVE @6obj$ @5, ADDMISSIONOBJECTIVE @7obj$ @5,
				SETVAR quad@5Target$ 0


		// Ex: captain_Grey owns Greenobj1, then Greenobj1 is picked up.
		IF PICKEDUP @1obj$ TESTVAR quad@6Target$ = 6 THEN
			TRIGGER syncclientB_@1_@5_@6_@7_$

			IF syncclientB_@1_@5_@6_@7_$ THEN
				ADDVAR @6score -1,
				ADDMISSIONOBJECTIVE @1obj$ @6, ADDMISSIONOBJECTIVE @5obj$ @6, ADDMISSIONOBJECTIVE @7obj$ @6,
				SETVAR quad@6Target$ 0


		// Ex: captain_Blue owns Greenobj1, then Greenobj1 is picked up.
		IF PICKEDUP @1obj$ TESTVAR quad@7Target$ = 5 THEN
			TRIGGER syncclientC_@1_@5_@6_@7_$

			IF syncclientC_@1_@5_@6_@7_$ THEN
				ADDVAR @7score -1,
				ADDMISSIONOBJECTIVE @1obj$ @7, ADDMISSIONOBJECTIVE @5obj$ @7, ADDMISSIONOBJECTIVE @6obj$ @7,
				SETVAR quad@7Target$ 0
	}
}
EXPAND WinchableHandler ( Green 940 941 942 Tan Grey Blue   )
EXPAND WinchableHandler ( Tan   943 944 945 Grey Blue Green )
EXPAND WinchableHandler ( Grey  946 947 948 Blue Green Tan  )
EXPAND WinchableHandler ( Blue  949 950 951 Green Tan Grey  )
