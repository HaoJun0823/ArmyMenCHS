MACRO macrologplace 1
{
	IF PICKEDUP buildlog@1 THEN		//1 mean log is on copter
		SETVAR varlog1 1		//not sure if this is needed

	IF DROPPED buildlog@1 THEN		//0 means no longer has it
		SETVAR varlog1 0,
		SETVAR varloglocx@1 GETX me,
		SETVAR varloglocy@1 GETY me,
		TRIGGER rightspota@1,
		TRIGGER rightspotc@1

	IF rightspota@1 TESTVAR varloglocx@1 > 2300 THEN	//perfect placement (2511,4339) to (2333,4394)
		TRIGGER rightspotb@1

	IF rightspotb@1 TESTVAR varloglocx@1 < 2600 THEN
		//showmessage "原木 X 坐标正确",
		TRIGGER placea@1
	
	IF rightspotc@1 TESTVAR varloglocy@1 > 4300 THEN
		TRIGGER rightspotd@1

	IF rightspotd@1 TESTVAR varloglocy@1 < 4400 THEN
		//showmessage "原木 Y 坐标正确",
		TRIGGER placeb@1

	If count 2 of placea@1 placeb@1 THEN
		//showmessage "原木已放置！",
		TRIGGER logplop@1
	
}
