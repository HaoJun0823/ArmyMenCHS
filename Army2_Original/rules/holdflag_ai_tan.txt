// this gets loaded on the host if the tan army is AI controlled

// This variable holds the current action for tan
// 0 means no action, 1 means follow black flag, 2 means get powerups, 3 means hold the flag
variable tanAction 0

if startup1 then
	triggerdelay 1000 tanChooseNewAction,
	triggerdelay 2000 tanRegularOrders

if tanChooseNewAction then
	setvar tanAction 0,
	trigger tanCheckIfHoldFlag,
	trigger tanCheckIfFlagUnheld,
	trigger tanChoseAction

// if it's time to choose a new action but we are currently owning the hill then
// we should always stay on the hill

if tanCheckIfHoldFlag testvar hasitem tansarge1 blackflag1 = 1 then
	trigger tanReturnToBase
	
// if it's time to choose a new action and the hill is not owned then head straight for it

if tanCheckIfFlagUnheld testvar hasitem greensarge1 blackflag1 = 0 and hasitem bluesarge1 blackflag1 = 0 and hasitem greysarge1 blackflag1 = 0 then
	trigger tanFollowFlag
	
// if tan has no current action then pick an action
// ideally we would base this on how well armed we are
if tanChooseAction testvar tanAction = 0 then
	trigger tanPickRandomAction,
	triggerdelay 1000 tanChooseAction

if tanPickRandomAction then random
	trigger tanFollowFlag,
	trigger tanGetPowerUps

// these events setup the tanAction variable and do the actions

if tanFollowFlag then
	setvar tanAction 1,
	order tansarge1 follow blackflag1 inmode attack,
	order tan group 1 follow tansarge1 inmode defend,
	order tan group 2 follow blackflag1 inmode attack,
	order tan group 3 follow blackflag1 inmode attack,
	trigger tanStartTimeOut

if tanGetPowerUps then
	setvar tanAction 2,
	order tansarge1 follow blackflag1 inmode attack,
	order tan group 1 follow tansarge1 inmode defend,
	order tan group 2 follow blackflag1 inmode attack,
	order tan group 3 follow blackflag1 inmode attack,
	trigger tanStartTimeOut

if tanReturnToBase then
	setvar tanAction 3,
	order tansarge1 goto tanbase inmode attack,
	order tan group 1 follow tansarge1 inmode defend,
	order tan group 2 goto tanbase inmode defend,
	order tan group 3 goto tanbase inmode defend,
	trigger tanStartTimeOut

if tanHoldFlag then
	setvar tanAction 4,
	order tansarge1 goto tanbase inmode attack,
	order tan group 1 follow tansarge1 inmode defend,
	order tan group 2 goto tanbase inmode defend,
	order tan group 3 goto tanbase inmode defend,
	trigger tanStartTimeOut

// these events chose a new action if we get on the hill

if pickedup blackflag1 by tansarge1 then
	trigger tanReturnToBase

if pickedup blackflag1 by greensarge1 then
	trigger tanFollowFlag

if pickedup blackflag1 by bluesarge1 then
	trigger tanFollowFlag

if pickedup blackflag1 by greysarge1 then
	trigger tanFollowFlag

if padon tanbase testvar hasitem tansarge1 blackflag1 = 1 then
    trigger tanHoldFlag

// these events choose new actions if a sarge is killed

if killed tansarge1 then
	triggerdelay 5000 tanGetPowerups

// these are the standing order regardless of what mode sarge is in
	
if tanRegularOrders then
	order tan group 0 follow tansarge1 inmode attack,
	setaimode tan group 4 defend,
	setaimode tan group 5 defend,
	order tan group 7 goto tanbase inmode evade,
	triggerdelay 5000 tanRegularOrders

// timeout code - if mode has not changed for a while then change it

variable tanTimeOutVar 0
variable tanLastTimeOutVar 1

if tanStartTimeOut then
    addvar tanTimeOutVar 1,
	triggerdelay 120000 tanCheckTimeOut

if tanCheckTimeOut testvar tanTimeOutVar = tanLastTimeOutVar then
    trigger tanChooseNewAction

if tanCheckTimeOut testvar tanTimeOutVar <> tanLastTimeOutVar then
	addvar tanLastTimeOutVar 1
