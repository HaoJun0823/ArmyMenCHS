variable greenFlagLastTimeOutVar 1

	triggerdelay 120000 greenFlagCheckTimeOut

    addvar greenFlagTimeOutVar 1

if greenFlagCheckTimeOut testvar greenFlagTimeOutVar = greenFlagLastTimeOutVar then
	showmessage "Green Flag was dropped and not picked up for 2 minutes - moved to base",
	createexplosion greenflag1 128 0 green,
	playsound "mp_rebaseflag.wav" (0,0) 2 1,
    moveitem greenflag1 gflagbase,
	triggerdelay 1000 greenFlagLastTimeOutIncr

if greenFlagCheckTimeOut testvar greenFlagTimeOutVar <> greenFlagLastTimeOutVar then
	triggerdelay 1000 greenFlagLastTimeOutIncr

if greenFlagLastTimeOutIncr then
	addvar greenFlagLastTimeOutVar 1
