variable ownHillBonus 10

variable greenonhill 0

variable blueonhill 0
variable greyonhill 0

// 0 = no owner, 1 = green, 2 = tan, 3 = blue, 4 = grey
variable hillowner 0
variable oldhillowner 0

// these pairs of variables are used to prevent sarge leaving and returning to the hill within
// 1 second and scoring at a double rate
variable greenGainsHillCount 0
variable greenKeepsHillCheck 0
variable tanGainsHillCount 0
variable tanKeepsHillCheck 0
variable blueGainsHillCount 0
variable blueKeepsHillCheck 0
variable greyGainsHillCount 0
variable greyKeepsHillCheck 0

pad tanhill 20 tansarge1
pad bluehill 20 bluesarge1
pad greyhill 20 greysarge1

// check for green sarge owning the hill

if padon greenhill then
	setvar greenonhill 1,
    trigger checkgreenownshill

if padoff greenhill then
	setvar greenonhill 0,
	trigger greenoffhill

if killed greensarge1 then
	setvar greenonhill 0,
	trigger greenoffhill

if greenoffhill testvar hillowner = 1 then
	setvar hillowner 0,
	trigger hillOwnerChanged,
	trigger checktanownshill,
	trigger checkblueownshill,
	trigger checkgreyownshill

if checkgreenownshill testvar greenonhill = 1 and hillowner = 0 then
	setvar hillowner 1,
	trigger hillOwnerChanged,
    showmessage "Green is king of the hill",
	addvar greenGainsHillCount 1,
	triggerdelay 3000 checkgreenscoresinitial

if checkgreenscoresinitial then
	addvar greenKeepsHillCheck 1

if checkgreenscoresinitial testvar hillowner = 1 then
	trigger checkgreenscores

if checkgreenscores testvar hillowner = 1 and greenKeepsHillCheck = greenGainsHillCount then
	addvar greenscore ownHillBonus,
	playsound "mp_tick.wav" (0,0) 2 3,
	trigger checkgreenwins,
	triggerdelay 3000 checkgreenscores
