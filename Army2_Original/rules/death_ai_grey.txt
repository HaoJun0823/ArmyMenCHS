// This variable holds the current action for grey
// 0 means no action, 1 means attack green, 2 means attack blue, 3 means attack tan
variable greyAction 0

// this variable is 
// this gets loaded on the host if the grey army is AI controlled
if startup1 then
	triggerdelay 1000 greyChooseNewAction,
	triggerdelay 2000 greyRegularOrders

if greyChooseNewAction then
	setvar greyAction 0,
	trigger greyChooseAction

// if grey has no current action then pick a random action
// keep doing this until an action is set
if greyChooseAction testvar greyAction = 0 then
	trigger greyPickRandomAction,
	triggerdelay 1000 greyChooseAction

if greyPickRandomAction then random
	trigger greySetAttackGreen,
	trigger greySetAttackBlue,
	trigger greySetAttackTan

// these events setup the greyAction variable

// before attacking green make sure green is in the game and is not an ally
if greySetAttackGreen testvar IsColorInGame green > 0 and IsAlly grey green = 0 then
	setvar greyAction 1,
	trigger greyAttackGreen,
	trigger startTimeOut

// before attacking blue make sure blue is in the game and is not an ally
if greySetAttackBlue testvar IsColorInGame blue > 0 and IsAlly grey blue = 0 then
	setvar greyAction 2,
	trigger greyAttackBlue,
	trigger startTimeOut

// before attacking tan make sure tan is in the game and is not an ally
if greySetAttackTan testvar IsColorInGame tan > 0 and IsAlly grey tan = 0 then
	setvar greyAction 3,
	trigger greyAttackTan,
	trigger startTimeOut

// these events do the actual attack orders

if greyAttackGreen testvar greyAction = 1 then
	order greysarge1 follow greensarge1 inmode attack,
	order grey group 1 follow greysarge1 inmode defend,
	order grey group 2 follow greensarge1 inmode attack,
	order grey group 3 follow greensarge1 inmode attack,
	order grey group 4 follow greensarge1 inmode attack,
	order grey group 5 follow greensarge1 inmode attack,
	order grey group 0 follow greensarge1 inmode attack,
	triggerdelay 500 greyAttackGreen

if greyAttackBlue testvar greyAction = 2 then
	order greysarge1 follow bluesarge1 inmode attack,
	order grey group 1 follow greysarge1 inmode defend,
	order grey group 2 follow bluesarge1 inmode attack,
	order grey group 3 follow bluesarge1 inmode attack,
	order grey group 4 follow bluesarge1 inmode attack,
	order grey group 5 follow bluesarge1 inmode attack,
	order grey group 0 follow bluesarge1 inmode attack,
	triggerdelay 500 greyAttackBlue

if greyAttackTan testvar greyAction = 3 then
	order greysarge1 follow tansarge1 inmode attack,
	order grey group 1 follow greysarge1 inmode defend,
	order grey group 2 follow tansarge1 inmode attack,
	order grey group 3 follow tansarge1 inmode attack,
	order grey group 4 follow tansarge1 inmode attack,
	order grey group 5 follow tansarge1 inmode attack,
	order grey group 0 follow tansarge1 inmode attack,
	triggerdelay 500 greyAttackTan

// these events choose new actions if a sarge is killed

if killed sarge by grey then
	triggerdelay 5000 greyChooseNewAction

if killed greysarge1 then
	triggerdelay 5000 greyChooseNewAction

// these are the sgreyding order regardless of what mode sarge is in
	
if greyRegularOrders then
	order grey group 7 goto greybase inmode evade,
	triggerdelay 5000 greyRegularOrders

// timeout code - if mode has not changed for a while then change it

variable greyTimeOutVar 0
variable greyLastTimeOutVar 1

if greyStartTimeOut then
    addvar greyTimeOutVar 1,
	triggerdelay 120000 greyCheckTimeOut

if greyCheckTimeOut testvar greyTimeOutVar = greyLastTimeOutVar then
    trigger greyChooseNewAction

if greyCheckTimeOut testvar greyTimeOutVar <> greyLastTimeOutVar then
	addvar greyLastTimeOutVar 1
