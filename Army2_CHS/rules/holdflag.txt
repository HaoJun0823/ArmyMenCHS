//////////////////////////////////////////////////////////////////////////////
// Hold the flag specific section
//////////////////////////////////////////////////////////////////////////////

variable holdFlagBonus 10

// these pairs of variables are used to prevent sarge leaving and returning to the hill within
// 1 second and scoring at a double rate
variable greenPickupFlagCount 0
variable greenKeepsFlagCheck 0
variable tanPickupFlagCount 0
variable tanKeepsFlagCheck 0
variable bluePickupFlagCount 0
variable blueKeepsFlagCheck 0
variable greyPickupFlagCount 0
variable greyKeepsFlagCheck 0

// in holdflag the black flag is a strategic item
if startup1 then
	deploy bkflagbase,
	deploy blackflag1,
	setitemflag blackflag1 strategic 1

// check for green sarge holding the flag

if pickedup blackflag1 by greensarge1 then
    showmessage "Green holds the flag",
	playsound "mp_gotflag.wav" (0,0) 3 1,
	addvar greenPickupFlagCount 1,
    triggerdelay 3000 checkgreenholdsflag

if checkgreenholdsflag then
    addvar greenKeepsFlagCheck 1

if checkgreenholdsflag testvar hasitem greensarge1 blackflag1 = 1 then
	trigger checkgreenscores

if checkgreenscores testvar hasitem greensarge1 blackflag1 = 1 and greenKeepsFlagCheck = greenPickupFlagCount then
	addvar greenscore holdFlagBonus,
	playsound "mp_tick.wav" (0,0) 2 3,
	trigger checkgreenwins,
	triggerdelay 3000 checkgreenscores

// check for tan sarge holding the flag

if pickedup blackflag1 by tansarge1 then
    showmessage "Tan holds the flag",
	playsound "mp_gotflag.wav" (0,0) 3 1,
	addvar tanPickupFlagCount 1,
    triggerdelay 3000 checktanholdsflag

if checktanholdsflag then

if checktanholdsflag testvar hasitem tansarge1 blackflag1 = 1 then

// if flag is dropped and not picked up for 2 minutes then return it to it's base

variable flagTimeOutVar 0
variable flagLastTimeOutVar 1

if dropped blackflag1 then
	triggerdelay 120000 flagCheckTimeOut

if pickedup blackflag1 then
    addvar flagTimeOutVar 1

if flagCheckTimeOut testvar flagTimeOutVar = flagLastTimeOutVar then
	showmessage "Black Flag was dropped and not picked up for 2 minutes - moved to base",
	createexplosion blackflag1 128 0 green,
	playsound "mp_rebaseflag.wav" (0,0) 2 1,
    moveitem blackflag1 bkflagbase,
	triggerdelay 1000 flagLastTimeOutIncr

if flagCheckTimeOut testvar flagTimeOutVar <> flagLastTimeOutVar then
	triggerdelay 1000 flagLastTimeOutIncr

if flagLastTimeOutIncr then
	addvar flagLastTimeOutVar 1
